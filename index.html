// NestJS Memory-Safe Stream & Heap Profiling Interceptor
@Injectable()
export class OcrPipelineResilienceInterceptor implements NestInterceptor {
  intercept(ctx: ExecutionContext, next: CallHandler): Observable<any> {
    const t0 = performance.now(), heap0 = process.memoryUsage().heapUsed;
    return next.handle().pipe(
      timeout({ each: 8500 }),
      retry({ count: 2, delay: 400 }),
      finalize(() => this.metrics.recordLatencyAndHeap(performance.now() - t0, heap0))
    );
  }
}

-- BigQuery Partitioned & Clustered Drive Forecast Optimization
CREATE OR REPLACE TABLE `drive_prod.slot_forecasts_v3`
PARTITION BY DATE(slot_timestamp)
CLUSTER BY store_id, preparation_zone AS
SELECT store_id, preparation_zone, slot_timestamp,
  ROUND(AVG(orders_count) OVER(
    PARTITION BY store_id, EXTRACT(DAYOFWEEK FROM slot_timestamp)
    ORDER BY UNIX_DATE(DATE(slot_timestamp)) RANGE BETWEEN 28 PRECEDING AND 1 PRECEDING
  ) * seasonal_factor, 1) AS predicted_pickers
FROM `drive_prod.raw_transactions_stream` WHERE _PARTITIONDATE >= DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY);
